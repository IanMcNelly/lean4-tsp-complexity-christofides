    (s : SolverState n) : SolverState n
:=
  match s.status with
  | .accepted _ => s
  | .rejected => s
  | .searching =>
    match s.queue with
    | [] =>
      match s.bestTour with
      | some t => ⟨.accepted t, [], s.bestCost, s.bestTour, s.steps + 1⟩
      | none => ⟨.rejected, [], none, none, s.steps + 1⟩

        ⟨.accepted p, ps, some (tourCost c p), some p, s.steps + 1⟩
